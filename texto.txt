local espActivo = false

local function aplicarESP(modelo, nombre, color)
    if not modelo:FindFirstChild("HumanoidRootPart") then return end

    local hl = modelo:FindFirstChild("HondurasHL") or Instance.new("Highlight")
    hl.Name = "HondurasHL"
    hl.FillColor = color
    hl.OutlineColor = Color3.new(1, 1, 1)
    hl.FillTransparency = 0.5
    hl.Enabled = true
    hl.Parent = modelo

    local bg = modelo.HumanoidRootPart:FindFirstChild("HondurasTag") or Instance.new("BillboardGui", modelo.HumanoidRootPart)
    bg.Name = "HondurasTag"
    bg.Size = UDim2.new(0, 200, 0, 50)
    bg.StudsOffset = Vector3.new(0, 3.5, 0)
    bg.AlwaysOnTop = true
    bg.Enabled = true

    local tl = bg:FindFirstChild("Lbl") or Instance.new("TextLabel", bg)
    tl.Name = "Lbl"
    tl.Size = UDim2.new(1, 0, 1, 0)
    tl.BackgroundTransparency = 1
    tl.TextColor3 = color
    tl.TextStrokeTransparency = 0
    tl.Font = Enum.Font.GothamBold
    tl.TextSize = 13

    spawn(function()
        while modelo and modelo:FindFirstChild("HumanoidRootPart") and espActivo do
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                local dist = math.floor((player.Character.HumanoidRootPart.Position - modelo.HumanoidRootPart.Position).Magnitude)
                tl.Text = nombre .. " [" .. dist .. "m]"
            end
            task.wait(0.25)
        end
        pcall(function() bg:Destroy() end)
        pcall(function() hl:Destroy() end)
    end)
end

    espBtn.BackgroundColor3 = espActivo and Color3.fromRGB(60, 40, 10) or Color3.fromRGB(18, 18, 32)

    if espActivo then
        spawn(function()

                -- ESP Jugadores
                for _, v in pairs(Players:GetPlayers()) do
                    if v ~= player and v.Character then
                        aplicarESP(v.Character, v.Name, Color3.fromRGB(255, 60, 60))

                        aplicarESP(v, v.Name, Color3.fromRGB(0, 220, 255))

    end
end)

local Camera         = workspace.CurrentCamera
local aimbotActivo   = false
local macroXZActivo  = false
local aimMaxDist     = 500
local aimFOV         = 150

local function getClosestPlayerAim()
    local target = nil
    local shortest = aimFOV
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = p.Character.HumanoidRootPart
            local myHRP = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
            if not myHRP then continue end
            local dist = (myHRP.Position - hrp.Position).Magnitude
            if dist <= aimMaxDist then
                local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                if onScreen then
                    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
                    local distCenter = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    if distCenter < shortest then
                        target = hrp
                        shortest = distCenter
                    end
                end
            end
        end
    end
    return target
end

-- Bucle Aimbot (RenderStepped)
RunService.RenderStepped:Connect(function()
    local isShiftLock = UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter
    if aimbotActivo and isShiftLock then
        local target = getClosestPlayerAim()
        if target then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, target.Position)
        end
    end
end)

-- =====================================================
--                   KILL AURA
-- =====================================================
-- Usa los RemoteEvents reales de Blox Fruits para registrar
-- hits en rango. El botón es arrastrable y NO toca el input
-- de cámara (InputChanged solo mueve el botón, no interfiere).

local kaActivo      = false
local kaRange       = 5000
local kaConnection  = nil

-- Remotes de Blox Fruits
local Net            = pcall(function() return ReplicatedStorage.Modules.Net end) and ReplicatedStorage:FindFirstChild("Modules") and ReplicatedStorage.Modules:FindFirstChild("Net")
local RegisterHit    = Net and Net:FindFirstChild("RE/RegisterHit")
local RegisterAttack = Net and Net:FindFirstChild("RE/RegisterAttack")

local function kaAtacarTargets(targets)
    pcall(function()
        if not targets or #targets == 0 then return end
        local allTargets = {}
        for _, char in pairs(targets) do
            local head = char:FindFirstChild("Head")
            if head then
                allTargets[#allTargets + 1] = { char, head }
            end
        end
        if #allTargets == 0 then return end
        if RegisterAttack then RegisterAttack:FireServer(0) end
        if RegisterHit    then RegisterHit:FireServer(allTargets[1][2], allTargets) end
    end)
end

local function kaStart()
    if kaConnection then task.cancel(kaConnection) end
    kaConnection = task.spawn(function()
        while kaActivo do
            task.wait(0.01)
            local myChar = player.Character
            local myHRP  = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if not myHRP then continue end

            local targets = {}

            -- Jugadores
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= player and p.Character then
                    local hum = p.Character:FindFirstChild("Humanoid")
                    local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                    if hum and hrp and hum.Health > 0 then
                        if (hrp.Position - myHRP.Position).Magnitude <= kaRange then
                            targets[#targets + 1] = p.Character
                        end
                    end
                end
            end

            -- NPCs (carpeta Enemies)
            local enemies = workspace:FindFirstChild("Enemies")
            if enemies then
                for _, npc in pairs(enemies:GetChildren()) do
                    local hum = npc:FindFirstChild("Humanoid")
                    local hrp = npc:FindFirstChild("HumanoidRootPart")
                    if hum and hrp and hum.Health > 0 then
                        if (hrp.Position - myHRP.Position).Magnitude <= kaRange then
                            targets[#targets + 1] = npc
                        end
                    end
                end
            end

            if #targets > 0 then
                kaAtacarTargets(targets)
            end
        end
    end)
end

local function kaStop()
    if kaConnection then
        task.cancel(kaConnection)
        kaConnection = nil
    end
end

killAuraBtn.MouseButton1Click:Connect(function()
    kaActivo = not kaActivo
    killAuraBtn.Text = kaActivo and "✅  Kill Aura: ON" or "☠️  Kill Aura"
    killAuraBtn.BackgroundColor3 = kaActivo and Color3.fromRGB(35, 0, 55) or Color3.fromRGB(18, 18, 32)
    if kaActivo then kaStart() else kaStop() end
end)
